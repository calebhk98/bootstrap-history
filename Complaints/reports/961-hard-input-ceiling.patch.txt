+"""The most a producer bids for an input its run cannot go ahead without (Complaint 961): what the run can pay for it
+and still pay, not a multiple of the input's price, so the bound does not ratchet with the price bid."""
+
+
+def hard_input_ceiling(price: float, per_run: float, margin_for_input: float) -> float:
+    """The price plus the margin left for this input per unit of it a run uses: at that price the run just pays."""
+    return price + max(0.0, margin_for_input) / per_run
+
+
+def working_capital_charge(variable_cost_per_run: float, interest_rate: float) -> float:
+    """Interest on a year's variable cost a run pays before its sales come in."""
+    return max(0.0, interest_rate) * variable_cost_per_run
+
+
+def margin_share(margin: float, hard_inputs_short: int) -> float:
+    """What each of the hard inputs bought short may take of the run's margin: all of it for one, an equal part of it
+    for several, so their ceilings together never ask more than the run can pay."""
+    return max(0.0, margin) / max(1, hard_inputs_short)

+"""Complaint 961: a producer's bid for a hard input (one without which its run cannot go ahead, with no substitute)
+stops at the most the run can pay for it: the run's revenue less the other inputs, labour, rent and the interest on the
+working capital it ties up, per unit of this input. It does not follow the input's own price."""
+
+QUICK_TOPIC = True
+
+import unittest
+
+from sim.economy import hard_input_ceiling
+
+
+class HardInputCeilingTests(unittest.TestCase):
+    def test_the_ceiling_is_the_price_plus_the_margin_left_per_unit_of_the_input(self):
+        # a run uses 0.5 of the input at 2; the margin left after every cost is 3 a run: 2 + 3 / 0.5
+        self.assertAlmostEqual(hard_input_ceiling.hard_input_ceiling(price=2.0, per_run=0.5, margin_for_input=3.0), 8.0)
+
+    def test_no_margin_left_leaves_the_price(self):
+        self.assertEqual(hard_input_ceiling.hard_input_ceiling(2.0, 0.5, -4.0), 2.0)
+
+    def test_the_most_the_run_can_pay_does_not_move_with_the_input_price(self):
+        # revenue 100, other costs 40, the input 0.5 a run: at any price the run pays up to (100 - 40) / 0.5
+        for price in (1.0, 20.0, 119.0):
+            margin = 100.0 - 40.0 - 0.5 * price
+            self.assertAlmostEqual(hard_input_ceiling.hard_input_ceiling(price, 0.5, margin), 120.0)
+
+    def test_working_capital_costs_the_interest_on_a_years_variable_cost(self):
+        self.assertAlmostEqual(hard_input_ceiling.working_capital_charge(variable_cost_per_run=40.0, interest_rate=0.05), 2.0)
+        self.assertEqual(hard_input_ceiling.working_capital_charge(40.0, -0.02), 0.0)
+
+    def test_the_margin_is_shared_among_the_hard_inputs_bought_short(self):
+        self.assertAlmostEqual(hard_input_ceiling.margin_share(margin=6.0, hard_inputs_short=3), 2.0)
+        self.assertEqual(hard_input_ceiling.margin_share(6.0, 0), 6.0)
+        self.assertEqual(hard_input_ceiling.margin_share(-6.0, 2), 0.0)
+
+    def test_the_ceilings_of_inputs_bought_together_add_up_to_no_more_than_the_run_can_pay(self):
+        margin = 6.0
+        share = hard_input_ceiling.margin_share(margin, 2)
+        cost = sum((hard_input_ceiling.hard_input_ceiling(price, per_run, share) - price) * per_run
+                   for price, per_run in ((1.0, 2.0), (5.0, 0.1)))
+        self.assertAlmostEqual(cost, margin)
+
+
+if __name__ == "__main__":
+    unittest.main()
