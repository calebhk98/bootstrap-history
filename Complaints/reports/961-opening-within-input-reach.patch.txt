+"""Sizing the opening's producers to the inputs in their reach.
+
+A run stops when a hard input is short (producers.produce is Leontief), and in the year an input comes from the
+market area of that good the producer stands in. The placement sizes each recipe to national runs, so an area
+whose users outweigh its makers (an input only some tiles can grow, a maker placed where its output is wanted)
+plans runs its inputs cannot feed. `reachable_shares` finds, for placed producers, the share of its runs each can
+work with the inputs its area's makers supply (themselves cut to what their own inputs allow). `shifted_runs` first
+moves the runs a recipe cannot work where it stands to its producers elsewhere whose areas have the inputs to spare
+(a processor settles where its inputs are, its output carried to where it is wanted), and `fit_to_reach` sets the
+producers' capacity and expected sales to the shifted runs, cut to what is still out of reach.
+"""
+from typing import Callable, Dict, Mapping, NamedTuple, Set, Tuple
+
+from .replaced import replaced
+from .types import GoodId, Recipe, TileId
+
+REACH_PASSES = 40           # numerical bound on cutting users to their area's supply (a chain of makers settles stage by stage)
+REACH_TOLERANCE = 1e-6      # a numerical bound: a share this close to its last is settled
+SHIFT_ROUNDS = 6            # numerical bound on moving runs to where inputs are spare (a move uses up the spare it fills)
+
+AreaOf = Callable[[GoodId, TileId], str]
+Balance = Dict[Tuple[GoodId, str], float]
+
+
+class Worked(NamedTuple):
+    """The runs a placed producer works and the output share a run yields."""
+    recipe_id: str
+    tile: TileId
+    runs: float
+    yield_factor: float
+
+
+def _makers(workers: Mapping[str, Worked], recipes: Mapping[str, Recipe]) -> Set[GoodId]:
+    return {good for worker in workers.values() for good, quantity in recipes[worker.recipe_id].outputs.items() if quantity > 0.0}
+
+
+def _balances(workers: Mapping[str, Worked], share: Mapping[str, float], recipes: Mapping[str, Recipe],
+              area_of: AreaOf, makers: Set[GoodId]) -> Tuple[Balance, Balance]:
+    """(supply, need) of each (good, market area) at the runs `workers` work times `share`: what their recipes make, and
+    what they take of the hard inputs some placed producer makes."""
+    supply: Balance = {}
+    need: Balance = {}
+    for producer_id, worker in sorted(workers.items()):
+        recipe = recipes[worker.recipe_id]
+        worked = worker.runs * share[producer_id]
+        for good, quantity in recipe.outputs.items():
+            if quantity > 0.0:
+                key = (good, area_of(good, worker.tile))
+                supply[key] = supply.get(key, 0.0) + worked * worker.yield_factor * quantity
+        for good, quantity in recipe.hard_inputs.items():
+            if quantity > 0.0 and good in makers:
+                key = (good, area_of(good, worker.tile))
+                need[key] = need.get(key, 0.0) + worked * quantity
+    return supply, need
+
+
+def reachable_shares(workers: Mapping[str, Worked], recipes: Mapping[str, Recipe], area_of: AreaOf) -> Dict[str, float]:
+    """{producer: share of its runs its inputs in reach allow}, 1 for a producer whose inputs are all there. A hard input
+    is counted in a market area only when some placed producer makes it (what nothing makes is stock, not supply). An
+    area's users are cut in proportion to its supply over their need, and a maker cut by its own inputs supplies less."""
+    makers = _makers(workers, recipes)
+    share = {producer_id: 1.0 for producer_id in workers}
+    for _pass in range(REACH_PASSES):
+        supply, need = _balances(workers, share, recipes, area_of, makers)
+        moved = 0.0
+        for producer_id, worker in sorted(workers.items()):
+            factor = 1.0
+            for good, quantity in recipes[worker.recipe_id].hard_inputs.items():
+                key = (good, area_of(good, worker.tile))
+                if quantity > 0.0 and good in makers and need.get(key, 0.0) > supply.get(key, 0.0):
+                    factor = min(factor, max(0.0, supply.get(key, 0.0)) / need[key])
+            if factor < 1.0:
+                moved = max(moved, share[producer_id] * (1.0 - factor))
+                share[producer_id] *= factor
+        if moved <= REACH_TOLERANCE:
+            break
+    return share
+
+
+def shifted_runs(workers: Mapping[str, Worked], recipes: Mapping[str, Recipe], area_of: AreaOf,
+                 spare_hours: Mapping[TileId, float], movable: Set[str]) -> Dict[str, float]:
+    """{producer: runs} after the runs a `movable` recipe's producers cannot work (`reachable_shares`) are moved to the
+    recipe's other producers whose areas have every input to spare, as many as their inputs and the hours `spare_hours`
+    leaves on their tiles allow. The total of a recipe's runs is kept or, where nothing can take them, left as it was."""
+    runs = {producer_id: worker.runs for producer_id, worker in workers.items()}
+    makers = _makers(workers, recipes)
+    spare = dict(spare_hours)
+    for _round in range(SHIFT_ROUNDS):
+        current = {producer_id: worker._replace(runs=runs[producer_id]) for producer_id, worker in workers.items()}
+        share = reachable_shares(current, recipes, area_of)
+        supply, need = _balances(current, share, recipes, area_of, makers)
+        room: Balance = {key: supply[key] - need.get(key, 0.0) for key in supply}
+        moved = False
+        for recipe_id in sorted(movable):
+            recipe = recipes[recipe_id]
+            members = sorted(producer_id for producer_id, worker in current.items() if worker.recipe_id == recipe_id)
+            lost = sum(runs[producer_id] * (1.0 - share[producer_id]) for producer_id in members)
+            if lost <= 0.0:
+                continue
+            hours = sum(recipe.labour_hours.values())
+            takers = {}
+            for producer_id in members:
+                if share[producer_id] < 1.0:
+                    continue
+                tile = current[producer_id].tile
+                keys = [((good, area_of(good, tile)), quantity) for good, quantity in recipe.hard_inputs.items()
+                        if quantity > 0.0 and good in makers]
+                headroom = min([max(0.0, room.get(key, 0.0)) / quantity for key, quantity in keys]
+                               + [spare.get(tile, float("inf")) / hours if hours > 0.0 else float("inf")])
+                if headroom > 0.0 and headroom < float("inf"):
+                    takers[producer_id] = (headroom, keys)
+            if not takers:
+                continue
+            crowd: Dict[Tuple[GoodId, str], int] = {}
+            for _headroom, keys in takers.values():
+                for key, _quantity in keys:
+                    crowd[key] = crowd.get(key, 0) + 1
+            fair = {producer_id: min([max(0.0, room.get(key, 0.0)) / quantity / crowd[key] for key, quantity in keys]
+                                     + [headroom])
+                    for producer_id, (headroom, keys) in takers.items()}
+            moving = min(lost, sum(fair.values()))
+            if moving <= 0.0:
+                continue
+            for producer_id in sorted(fair):
+                added = moving * fair[producer_id] / sum(fair.values())
+                runs[producer_id] += added
+                tile = current[producer_id].tile
+                spare[tile] = spare.get(tile, float("inf")) - added * hours
+                for key, quantity in takers[producer_id][1]:
+                    room[key] = room.get(key, 0.0) - added * quantity
+            for producer_id in members:     # the cut producers give up the moved runs in proportion to what they could not work
+                if share[producer_id] < 1.0:
+                    runs[producer_id] = max(0.0, runs[producer_id] * (1.0 - moving * (1.0 - share[producer_id]) / lost))
+            moved = True
+        if not moved:
+            break
+    return runs
+
+
+def fit_to_reach(setup, record, area_map) -> None:
+    """Size every placed producer to the inputs in its reach: move what a movable recipe cannot work where it stands to
+    its producers whose areas have the inputs to spare (`shifted_runs`), then cut each to the share its inputs allow
+    (`reachable_shares`). Capacity and expected sales change by the same factor."""
+    workers = {producer_id: Worked(producer.recipe_id, producer.tile, producer.expected_sales, producer.yield_factor)
+               for producer_id, producer in sorted(record.producers.items())}
+    movable = {recipe_id for recipe_id, recipe in setup.recipes.items()
+               if not (recipe.service or recipe.site_bound or setup.land_per_run.get(recipe_id, 0.0) > 0.0)
+               and _carries(setup, recipe)}
+    runs = shifted_runs(workers, setup.recipes, area_map.area_of, _spare_hours(setup, record), movable)
+    placed = {producer_id: worker._replace(runs=runs[producer_id]) for producer_id, worker in workers.items()}
+    shares = reachable_shares(placed, setup.recipes, area_map.area_of)
+    for producer_id, worker in workers.items():
+        factor = (runs[producer_id] / worker.runs if worker.runs > 0.0 else 1.0) * shares[producer_id]
+        if factor != 1.0:
+            producer = record.producers[producer_id]
+            record.producers[producer_id] = replaced(
+                producer, capacity_runs=producer.capacity_runs * factor, expected_sales=producer.expected_sales * factor)
+
+
+def _carries(setup, recipe: Recipe) -> bool:
+    """Whether what the recipe mainly makes can be carried to where it is wanted (a service, or a good with no portable
+    form, is made where it is used)."""
+    outputs = [good for good, quantity in recipe.outputs.items() if quantity > 0.0]
+    return bool(outputs) and all(setup.specs[good].portable for good in outputs if good in setup.specs)
+
+
+def _spare_hours(setup, record) -> Dict[TileId, float]:
+    """Hours a year each tile's working people have left after the placed producers' expected runs."""
+    spare = {tile: people * setup.working_share * setup.working_hours_per_year
+             for tile, people in setup.opening_population_by_tile.items()}
+    for producer in record.producers.values():
+        hours = sum(setup.recipes[producer.recipe_id].labour_hours.values())
+        spare[producer.tile] = spare.get(producer.tile, 0.0) - producer.expected_sales * hours
+    return spare
