+"""The share of a producer's workplaces that run at an expected revenue.
+
+One producer stands for many workplaces. Their cost per run differs; the producer's variable cost is their mean. A workplace runs in the short run when revenue covers the cash it must pay out (inputs and hired labour);
+rent and the like are due whether it runs or not. The cheapest workplace pays out CHEAPEST_WORKPLACE_COST_SHARE of the
+mean's cash, so no workplace runs below that. Costs rise from that floor as a power of the share running, with the
+exponent that gives the sourced elasticity at break-even and a mean equal to the producer's cost; the share running
+is then a power of the revenue above the floor, gentle between the floor and full capacity.
+"""
+import math
+from typing import Optional
+
+from sim.constants import declare
+
+SHORT_RUN_SUPPLY_ELASTICITY = declare(
+    "SHORT_RUN_SUPPLY_ELASTICITY", 0.225, kind="temporary_heuristic",
+    unit="elasticity of a producer's yearly runs to the price of its output, where revenue equals the mean workplace's cost",
+    source="Haile, Kalkuhl and von Braun 2016, Amer. J. Agric. Econ. 98(1) 172-190 (abstract: own-price supply "
+           "elasticities of four staple crops range from about 0.05 to 0.40; the value is the midpoint, read as a range "
+           "only); Schneider 2012, Oxford Discussion Papers in Economic and Social History 97, Table 5 (short-run wheat, "
+           "barley, oats from about zero to 0.42 across places and periods). "
+           "Complaints/reports/research-short-run-supply-elasticity.md",
+    confidence="D", retire="Complaints/960-year-zero-wage-clears-on-a-knife-edge-and-a-held-opening-wage-loses-hours.md",
+    why="The sources are crop acreage studies; no source was found for crafts or mines and the same figure stands for "
+        "them. The midpoint of the range is a judgement.")
+CHEAPEST_WORKPLACE_COST_SHARE = declare(
+    "CHEAPEST_WORKPLACE_COST_SHARE", 0.5, kind="temporary_heuristic",
+    unit="cash cost of the cheapest workplace one producer stands for, as a share of the mean workplace's",
+    source=None, confidence="D",
+    retire="Complaints/960-year-zero-wage-clears-on-a-knife-edge-and-a-held-opening-wage-loses-hours.md",
+    why="A low elasticity at break-even needs a floor under the workplaces' costs, or the curve has a fat lower tail "
+        "and a producer earning a tenth of its cost still runs a large share of its plant. The floor could come from "
+        "the spread of land fertility and deposit grade across the area's tiles.")
+
+
+def _shape(variable_cost_per_run: float, cash_cost_per_run: Optional[float]):
+    """(floor, exponent, share running at break-even). The workplace costs, in units of the producer's cost, are
+    floor + (1 - floor) * (share / share_at_break_even) ^ (1 / exponent) for share up to one; the exponent makes the
+    elasticity at break-even the sourced one, and the share at break-even makes their mean one."""
+    cash_share = 1.0 if cash_cost_per_run is None else min(1.0, max(0.0, cash_cost_per_run / variable_cost_per_run))
+    floor = CHEAPEST_WORKPLACE_COST_SHARE * cash_share
+    exponent = SHORT_RUN_SUPPLY_ELASTICITY * (1.0 - floor)
+    return floor, exponent, (1.0 / exponent + 1.0) ** -exponent
+
+
+def share_at_break_even(cash_share: float) -> float:
+    """The share of workplaces running where revenue equals the producer's cost, when `cash_share` of that cost is cash."""
+    return _shape(1.0, cash_share)[2]
+
+
+def run_share(revenue_per_run: float, variable_cost_per_run: float, cash_cost_per_run: Optional[float]) -> float:
+    """Share of the workplaces that run. Costs rise with the share as a power of the revenue above the floor,
+    share = share_at_break_even * ((revenue - floor) / (cost - floor)) ^ exponent with exponent = elasticity * (1 - floor / cost),
+    which gives `SHORT_RUN_SUPPLY_ELASTICITY` at revenue equal to cost. `cash_cost_per_run` is the part of the variable
+    cost paid out in cash (None: all of it)."""
+    if variable_cost_per_run <= 0.0:
+        return 1.0
+    if revenue_per_run <= 0.0 or not math.isfinite(variable_cost_per_run):
+        return 0.0
+    ratio = revenue_per_run / variable_cost_per_run
+    if not ratio > 0.0:     # a revenue too small against the cost to compare (the ratio underflows)
+        return 0.0
+    floor, exponent, at_break_even = _shape(variable_cost_per_run, cash_cost_per_run)
+    if ratio <= floor:
+        return 0.0
+    return min(1.0, at_break_even * ((ratio - floor) / (1.0 - floor)) ** exponent)
+
+
+def mean_cost_ratio(revenue_per_run: float, variable_cost_per_run: float, cash_cost_per_run: Optional[float]) -> float:
+    """The average cost of the workplaces that run, against the producer's (the mean of all): one when none run or
+    all do. With the share a power of the revenue above the floor, the workplaces that run cost the floor plus a fixed
+    fraction of the revenue's height above it (the integral of the power law, 1 / (1 / exponent + 1))."""
+    if run_share(revenue_per_run, variable_cost_per_run, cash_cost_per_run) <= 0.0:
+        return 1.0
+    floor, exponent, at_break_even = _shape(variable_cost_per_run, cash_cost_per_run)
+    height = min(revenue_per_run / variable_cost_per_run,
+                 floor + (1.0 - floor) * at_break_even ** (-1.0 / exponent))     # at full capacity the cost stops rising
+    return floor + (height - floor) / (1.0 / exponent + 1.0)

+"""`run_share` and `mean_cost_ratio` (sim/economy/producer_run_share.py): the share of a producer's workplaces that run at a
+revenue, with a gentle response near break-even and none below the cash the run must pay out."""

+import unittest

+from sim.economy.producer_run_share import (CHEAPEST_WORKPLACE_COST_SHARE, SHORT_RUN_SUPPLY_ELASTICITY,

+QUICK_TOPIC = True

+class RunShare(unittest.TestCase):
+
+    def test_most_workplaces_run_where_revenue_equals_their_mean_cost(self):
+        self.assertAlmostEqual(run_share(10.0, 10.0, 10.0), share_at_break_even(1.0))
+        self.assertTrue(0.5 < share_at_break_even(1.0) < 1.0)
+
+    def test_every_workplace_runs_at_a_few_times_the_cost_but_not_at_the_cost(self):
+        self.assertEqual(run_share(8.0, 1.0, 1.0), 1.0)
+        self.assertLess(run_share(1.0, 1.0, 1.0), 1.0)
+
+    def test_the_response_to_price_at_break_even_is_the_sourced_elasticity(self):
+        step = 1e-6
+        low, high = run_share(1.0 - step, 1.0, 1.0), run_share(1.0 + step, 1.0, 1.0)
+        elasticity = (math.log(high) - math.log(low)) / (math.log(1.0 + step) - math.log(1.0 - step))
+        self.assertAlmostEqual(elasticity, SHORT_RUN_SUPPLY_ELASTICITY, places=4)
+
+    def test_the_response_at_break_even_holds_when_part_of_the_cost_is_not_cash(self):
+        step = 1e-6
+        low, high = run_share(1.0 - step, 1.0, 0.4), run_share(1.0 + step, 1.0, 0.4)
+        elasticity = (math.log(high) - math.log(low)) / (math.log(1.0 + step) - math.log(1.0 - step))
+        self.assertAlmostEqual(elasticity, SHORT_RUN_SUPPLY_ELASTICITY, places=4)
