+## Update: year zero clears at the opening wage and near the opening prices (branch opening-wage-year-zero)
+
+Two causes, measured by `python3 sim/simulator.py spin-up-trace --fixture stationary --years 10`. (1) The workers' ask
+started at the bare reservation wage, far under the contracts the opening seeds, so year zero's new hires cleared where
+the hours sold (which rise with the wage through the plot) meet the hours wanted, and the hours-weighted mean over held
+and newly hired hours (each year about half the held hours are separated and re-hired at the going offer) read as a wage
+well under the opening's. A market that opens holding hours now opens asking the contract (`asks.opening_scale`).
+(2) Producers offered their safety stock as well as the year's output, more than buyers' budgets cover, so the price fell
+to the stock's holding reservation; the safety stock is now held back (`inventory.safety_stock_kept`). Still open: the
+stationary fixture's hired hours fall over the years because households spend less than the opening's final demand and
+cash piles up in households (hh cash 3.6M at year 0 to 5.4M at year 7 by the probe in 473), so the wage and price level
+drift down from decade two.

+def opening_scale(contracts: Mapping[str, float], held: Mapping[str, float], reservation: float) -> float:
+    """The ask of a market that opens holding hours: the hours-weighted contract wage over the reservation wage. A
+    hired worker's reservation is reset to the wage he is paid (Eurace Household_receive_wage, Labour/
+    Household_Employee_Functions.c), so hours that open at a contract ask it; one with nothing held, or no reservation
+    to measure by, asks the reservation itself."""
+    hours = math.fsum(max(0.0, held.get(employer, 0.0)) for employer in contracts)
+    if hours <= 0.0 or reservation <= 0.0:
+        return 1.0
+    return math.fsum(max(0.0, held.get(employer, 0.0)) * wage for employer, wage in sorted(contracts.items())) / hours / reservation
+

+"""A market that opens holding hours at contracts opens with its workers asking what those contracts pay (a hired
+worker's reservation is reset to the wage, Eurace Household_receive_wage), not the bare reservation wage."""
+
+QUICK_TOPIC = True
+
+import unittest
+
+from sim.labour.market import asks
+from sim.labour.market.records import MarketState
+
+
+class OpeningAskFollowsContracts(unittest.TestCase):
+    def test_ask_is_the_hours_weighted_contract_over_the_reservation(self):
+        scale = asks.opening_scale({"farm": 1.0, "mill": 2.0}, {"farm": 300.0, "mill": 100.0}, 0.5)
+        self.assertAlmostEqual(scale, 1.25 / 0.5)
+
+    def test_no_hours_held_asks_the_reservation_itself(self):
+        self.assertEqual(asks.opening_scale({}, {}, 0.5), 1.0)
+        self.assertEqual(asks.opening_scale({"farm": 1.0}, {"farm": 0.0}, 0.5), 1.0)
+
+    def test_a_reservation_of_nothing_leaves_the_scale_at_one(self):
+        self.assertEqual(asks.opening_scale({"farm": 1.0}, {"farm": 10.0}, 0.0), 1.0)
+
+    def test_a_remembered_scale_is_used_over_the_opening_one(self):
+        state = MarketState()
+        state.asks = {"vale": {"digger": 0.8}}
+        state.contract_wages = {"vale": {"digger": {"farm": 1.0}}}
+        state.hired_hours = {"vale": {"digger": {"farm": 10.0}}}
+        self.assertEqual(asks.scale_of(state, "vale", "digger", 0.5), 0.8)
+
+    def test_an_unremembered_scale_starts_from_the_contracts_held(self):
+        state = MarketState()
+        state.contract_wages = {"vale": {"digger": {"farm": 1.0}}}
+        state.hired_hours = {"vale": {"digger": {"farm": 10.0}}}
+        self.assertEqual(asks.scale_of(state, "vale", "digger", 0.5), 2.0)
+        self.assertEqual(asks.scale_of(state, "vale", "digger"), 1.0)
+
+
+if __name__ == "__main__":
+    unittest.main()
