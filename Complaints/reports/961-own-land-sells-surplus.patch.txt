+"""What a household's own land grows beyond its floors, and sells (Complaints 551, 538).
+
+A household that holds land works it where an hour there is worth more than the wage (households_hours_offered):
+the floors first, then the crop that fetches most per hour at the prices of the market, on the land the floors
+leave. Output on fixed land rises with hours to the labour elasticity (world/land.py), the same curve
+labour_ask_floor.plot_return_per_hour reads, so the hours it keeps and the harvest it gets agree. What the floors
+do not eat is offered on the goods market like any seller's stock. Land is worked beyond the floors only where a
+hectare nets more than the tile's rent, since the holder could let it instead. Plain functions of built inputs.
+"""
+import dataclasses
+import math
+from typing import List, Mapping, Optional, Sequence, Tuple
+
+from sim.world.need_basket import PricedNeed
+from sim.world.shared_constants import LABOUR_OUTPUT_ELASTICITY
+
+from .households_cohort import Cohort
+from .households_own import Option
+from .inventory import holding_reservation
+from .types import EDGE_PRODUCTION, AgentId, GoodId, GoodsMove, Offer, Recipe
+
+FOR_SALE_PURPOSE = "grown for sale"
+
+
+@dataclasses.dataclass(frozen=True)
+class SurplusCrop:
+    recipe_id: str
+    good: GoodId
+    effect: float                # need units one unit of the good gives
+    hours_per_unit: float        # hours a need unit takes at fertility one
+    price_per_unit: float        # money a need unit of it fetches
+    hectares_per_hour: float     # land one hour of the recipe's work takes at its own intensity
+
+
+def best_surplus_crop(priced: Sequence[PricedNeed], options: Mapping[str, Sequence[Option]],
+                      hectares_per_hour: Mapping[str, float]) -> Optional[SurplusCrop]:
+    """The crop on the land beyond the floors: of each need that has a floor and a land-only recipe, the one whose
+    good fetches most money per hour of work. None where no such good has a price."""
+    best: Optional[Tuple[float, SurplusCrop]] = None
+    for need in priced:
+        rows = options.get(need.spec.need_id)
+        if need.spec.subsistence_per_person <= 0.0 or not rows:
+            continue
+        hours_per_unit, recipe_id, good, effect = rows[0]
+        per_hour = hectares_per_hour.get(recipe_id, 0.0)
+        price = next((price for each, price, _effect, _share in need.goods if each == good), None)
+        if hours_per_unit <= 0.0 or per_hour <= 0.0 or effect <= 0.0 or price is None or price <= 0.0:
+            continue
+        crop = SurplusCrop(recipe_id, good, effect, hours_per_unit, price / effect, per_hour)
+        value_per_hour = crop.price_per_unit / hours_per_unit
+        if best is None or value_per_hour > best[0]:
+            best = (value_per_hour, crop)
+    return best[1] if best else None
+
+
+def reference_hours_per_hectare(crop: SurplusCrop, fertility: float) -> float:
+    return 1.0 / (fertility * crop.hectares_per_hour) if fertility > 0.0 else math.inf
+
+
+def units_per_hectare(crop: SurplusCrop, fertility: float, hours_per_hectare: float,
+                      elasticity: float = LABOUR_OUTPUT_ELASTICITY) -> float:
+    """Need units a hectare gives with this many hours on it: the reference yield times hours over the
+    reference hours to the labour elasticity."""
+    if hours_per_hectare <= 0.0 or fertility <= 0.0:
+        return 0.0
+    reference_units = 1.0 / (crop.hectares_per_hour * crop.hours_per_unit)
+    return reference_units * (hours_per_hectare / reference_hours_per_hectare(crop, fertility)) ** elasticity
+
+
+def hours_the_land_takes(crop: SurplusCrop, hectares: float) -> float:
+    """Most hours the hectares take: a run's hours on each run's land."""
+    return max(0.0, hectares) / crop.hectares_per_hour
+
+
+def grow(crop: SurplusCrop, hectares: float, hours: float, fertility: float,
+         elasticity: float = LABOUR_OUTPUT_ELASTICITY) -> Tuple[float, float]:
+    """(quantity of the good grown, hours used) when `hours` are worked on `hectares`, no more than the land takes."""
+    used = min(max(0.0, hours), hours_the_land_takes(crop, hectares))
+    if used <= 0.0 or hectares <= 0.0:
+        return 0.0, 0.0
+    units = hectares * units_per_hectare(crop, fertility, used / hectares, elasticity)
+    return units / crop.effect, used
+
+
+def net_margin_per_hectare(crop: SurplusCrop, fertility: float, wage_per_hour: float,
+                           elasticity: float = LABOUR_OUTPUT_ELASTICITY) -> float:
+    """Most a hectare nets its worker a year: its output at the crop's price less the hours at the wage, at the hours
+    per hectare where the next hour adds the wage (the whole of the land's hours where the wage is not positive)."""
+    cap = hours_the_land_takes(crop, 1.0)
+    if fertility <= 0.0 or cap <= 0.0:
+        return 0.0
+    scale = crop.price_per_unit * units_per_hectare(crop, fertility, 1.0, elasticity)      # money at one hour a hectare
+    hours = cap if wage_per_hour <= 0.0 else min(cap, (elasticity * scale / wage_per_hour) ** (1.0 / (1.0 - elasticity)))
+    return crop.price_per_unit * units_per_hectare(crop, fertility, hours, elasticity) - wage_per_hour * hours
+
+
+def hectares_worked_beyond_floors(held: float, floors_take: float, crop: Optional[SurplusCrop], fertility: float,
+                                  wage_per_hour: float, rent_per_hectare: float) -> float:
+    """The hectares a holder works for sale beyond its floors: the land left once the floors have theirs, where a
+    hectare nets more than the rent it would let for; none without a crop or where the rent is higher."""
+    left = max(0.0, held - max(0.0, floors_take))
+    if crop is None or left <= 0.0 or net_margin_per_hectare(crop, fertility, wage_per_hour) <= max(0.0, rent_per_hectare):
+        return 0.0
+    return left
+
+
+def hectares_the_floors_took(hours_by_recipe: Mapping[str, float], recipes: Mapping[str, Recipe],
+                             land_per_run: Mapping[str, float]) -> float:
+    """Land the hours already worked for the floors take (each recipe's runs on each run's land)."""
+    total = 0.0
+    for recipe_id, hours in sorted(hours_by_recipe.items()):
+        hours_per_run = math.fsum(recipes[recipe_id].labour_hours.values())
+        if hours_per_run > 0.0:
+            total += hours / hours_per_run * land_per_run.get(recipe_id, 0.0)
+    return total
+
+
+def grow_for_sale(cohort: Cohort, crop: SurplusCrop, planned_hectares: float, held: float, floors_take: float,
+                  spare_hours: float, fertility: float) -> Tuple[Optional[GoodsMove], float, float, float]:
+    """(move booking the harvest, quantity, hours used, hectares worked): the spare hours worked on the land planned
+    for the surplus, no more than the floors left of what it holds. None where nothing was grown."""
+    hectares = min(max(0.0, planned_hectares), max(0.0, held - floors_take))
+    quantity, used = grow(crop, hectares, spare_hours, fertility)
+    if quantity <= 0.0:
+        return None, 0.0, 0.0, 0.0
+    return GoodsMove(EDGE_PRODUCTION, cohort.agent_id, crop.good, cohort.tile, quantity, FOR_SALE_PURPOSE), quantity, used, hectares
+
+
+def offers_for_sale(seller: AgentId, tile: str, area_of_good, quantity_by_good: Mapping[GoodId, float],
+                    expected_price, spoilage_of, interest_rate: float) -> List[Offer]:
+    """The surplus put on its market at what carrying it to next year would net (a perishable sells at any price)."""
+    offers = []
+    for good, quantity in sorted(quantity_by_good.items()):
+        if quantity <= 0.0:
+            continue
+        reservation = holding_reservation(expected_price(good) or 0.0, interest_rate, spoilage_of(good), 0.0)
+        offers.append(Offer(seller, good, area_of_good(good), tile, quantity, reservation))
+    return offers
