+"""What a merchant learns from a market where buyers bid and nobody offered.
+
+Buyers' standing bids are posted where a merchant can see them. A market that cleared nothing for want of any
+seller still shows the most a buyer would pay and how much buyers wanted, so a merchant sees a destination worth
+carrying to, in the price and the volume it expects there.
+"""
+from typing import Optional, Sequence, Tuple
+
+from .market_quote import best_unfilled_bid
+from .types import Bid
+
+
+def unfilled_demand(bids: Sequence[Bid], traded: float, offered: float) -> Optional[Tuple[float, float]]:
+    """(best unfilled bid, quantity wanted at it) for a market where nothing traded and nothing was offered;
+    None otherwise, or when no buyer with money names a finite ceiling."""
+    if traded > 0.0 or offered > 0.0:
+        return None
+    ceiling = best_unfilled_bid(bids)
+    if ceiling is None:
+        return None
+    wanted = sum(bid.floor_quantity + bid.flexible_quantity for bid in bids
+                 if bid.budget > 0.0 and bid.maximum_price > 0.0)
+    return (ceiling, wanted) if wanted > 0.0 else None

+"""Complaint 644: a merchant sees the bids standing in a market where nothing was offered, as the price and the
+volume the market would take, so a surplus area elsewhere can draw a route to it."""
+
+QUICK_TOPIC = True
+
+import unittest
+
+from sim.economy.merchant_news import unfilled_demand
+from sim.economy.types import Bid
+
+
+def bid(floor_quantity, maximum_price, budget=1e9):
+    return Bid("buyer", "lead_kg", "area", "tile", floor_quantity, 0.0, 1.0, 0.0, budget, 0, maximum_price)
+
+
+class UnfilledDemandTests(unittest.TestCase):
+    def test_bids_with_no_offer_give_the_best_ceiling_and_the_quantity_wanted(self):
+        price, volume = unfilled_demand([bid(100.0, 1.5), bid(50.0, 1.8)], traded=0.0, offered=0.0)
+        self.assertEqual(price, 1.8)
+        self.assertEqual(volume, 150.0)
+
+    def test_a_market_that_traded_teaches_nothing_extra(self):
+        self.assertIsNone(unfilled_demand([bid(100.0, 1.5)], traded=10.0, offered=10.0))
+
+    def test_a_market_with_offers_unsold_teaches_nothing_extra(self):
+        self.assertIsNone(unfilled_demand([bid(100.0, 1.5)], traded=0.0, offered=40.0))
+
+    def test_a_bid_without_a_ceiling_or_cash_is_not_demand(self):
+        self.assertIsNone(unfilled_demand([bid(100.0, float("inf")), bid(100.0, 2.0, budget=0.0)], traded=0.0, offered=0.0))
+
+
+if __name__ == "__main__":
+    unittest.main()
