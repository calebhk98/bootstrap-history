Subject: [PATCH] Producers plan runs no further than their hard inputs reach

A producer holds its runs to what it holds plus what it can expect to buy: last year's fill where its order fell short, nothing where the market has nothing on offer, no limit where the order was filled in full. Labour and input bids follow the capped runs.

+"""How many runs a producer's inputs can support: what it holds plus what it can expect to buy this year.
+
+An employer does not take on hands for work it has no material for. The expected purchase of a hard input is
+read from what the producer actually got last year against its order (`short_fills`, kept on the producer) and
+from whether anything is on offer in its market area now. An order filled in full leaves no record: the input
+did not bind, so it caps nothing and the plan can grow again. Nothing here names a good or a trade.
+"""
+import math
+from typing import Callable, Dict, Mapping, Optional
+
+# An order counts as filled in full when the fill is within this share of it (rounding in settlement).
+FILLED_IN_FULL_TOLERANCE = 1e-6
+
+
+def short_fills(ordered: Mapping[str, float], got: Mapping[str, float]) -> Dict[str, float]:
+    """Of the goods the producer ordered, the quantity it got for each that was not filled in full. A good
+    filled in full, or not ordered, is absent: it capped nothing."""
+    return {good: got.get(good, 0.0) for good, quantity in ordered.items()
+            if quantity > 0.0 and got.get(good, 0.0) < quantity * (1.0 - FILLED_IN_FULL_TOLERANCE)}
+
+
+def expected_purchases(goods, on_offer: Callable[[str], bool], got_when_short: Mapping[str, float]) -> Dict[str, float]:
+    """Units of each good the producer can expect to buy this year: none where the market has nothing on offer,
+    last year's fill where that fell short of its order, no limit (infinity) otherwise."""
+    expected = {}
+    for good in goods:
+        if not on_offer(good):
+            expected[good] = 0.0
+        else:
+            expected[good] = got_when_short.get(good, math.inf)
+    return expected
+
+
+def runs_within_reach(hard_inputs: Mapping[str, float], held: Mapping[str, float],
+                      expected_purchase: Mapping[str, float]) -> float:
+    """The most runs the hard inputs support: for each, what is held plus what is expected to be bought, over
+    the units a run uses; the scarcest sets it. Infinity for a recipe with no hard input."""
+    reach = math.inf
+    for good, per_run in hard_inputs.items():
+        if per_run <= 0.0:
+            continue
+        units = held.get(good, 0.0) + expected_purchase.get(good, math.inf)
+        reach = min(reach, units / per_run)
+    return reach
+
+
+def runs_for_inputs(producer, hard_inputs: Mapping[str, float], view, planned_runs: float) -> float:
+    """`planned_runs` held to what the producer's hard inputs support."""
+    if not hard_inputs:
+        return planned_runs
+    area_of = view.area_of
+    offered = getattr(view, "offered_price", None)
+
+    def on_offer(good: str) -> bool:
+        return offered is None or offered(good, area_of(good, producer.tile)) is not None
+
+    held = {good: view.stock(producer.agent_id, good, producer.tile) for good in hard_inputs}
+    expected = expected_purchases(hard_inputs, on_offer, dict(producer.input_got_when_short))
+    return min(planned_runs, max(0.0, runs_within_reach(hard_inputs, held, expected)))
+
+
+def ordered_by_buyer(bids_by_market) -> Dict[str, Dict[str, float]]:
+    """The year's orders as buyer -> good -> quantity ordered (the bids' floors, the plan's inputs net of stock)."""
+    ordered: Dict[str, Dict[str, float]] = {}
+    for (good, _area), market_bids in bids_by_market.items():
+        for bid in market_bids:
+            if bid.floor_quantity > 0.0:
+                by_good = ordered.setdefault(bid.buyer, {})
+                by_good[good] = by_good.get(good, 0.0) + bid.floor_quantity
+    return ordered
+
+
+def got_when_short_after_year(hard_inputs, ordered: Mapping[str, float], fills) -> tuple:
+    """What the producer's record of short fills becomes at the year's close: (good, quantity got) pairs, sorted.
+    `ordered` is what it asked for by good; `fills` its buy fills."""
+    got: Dict[str, float] = {}
+    for fill in fills:
+        if fill.good in hard_inputs:
+            got[fill.good] = got.get(fill.good, 0.0) + fill.quantity
+    hard_orders = {good: quantity for good, quantity in ordered.items() if good in hard_inputs}
+    return tuple(sorted(short_fills(hard_orders, got).items()))
